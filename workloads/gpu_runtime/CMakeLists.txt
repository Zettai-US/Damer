cmake_minimum_required(VERSION 3.24)
# Build independently of LLVM/CIRCT and the x86 CXL lock implementation.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 121)
endif()
project(DamerGPU LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
  message(FATAL_ERROR "GB10 builds require CUDA 13 or newer; select its nvcc explicitly")
endif()
# Let the selected CUDA compiler supply its own runtime. FindCUDAToolkit can
# accidentally select a system CUDA 12 libcudart beside a CUDA 13 compiler.
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
add_executable(damer_gpu main.cu)
target_compile_options(damer_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME gpu_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_runtime.py $<TARGET_FILE:damer_gpu>)
set_tests_properties(gpu_contract PROPERTIES TIMEOUT 120)

option(DAMER_NCCL "Build the optional MPI/NCCL baseline" OFF)
if(DAMER_NCCL)
  find_package(MPI REQUIRED COMPONENTS CXX)
  find_path(NCCL_INCLUDE_DIR nccl.h REQUIRED)
  find_library(NCCL_LIBRARY nccl REQUIRED)
  add_executable(damer_nccl nccl_baseline.cu)
  target_include_directories(damer_nccl PRIVATE ${NCCL_INCLUDE_DIR})
  target_link_libraries(damer_nccl PRIVATE MPI::MPI_CXX ${NCCL_LIBRARY})
endif()
